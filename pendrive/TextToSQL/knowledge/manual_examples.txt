Q: Create a table Project whose Dept_ID is a foreign key to Department
K: create table primary key foreign key references not null data type

  Project_ID INT PRIMARY KEY,
  Project_Name VARCHAR(100) NOT NULL,
  Budget_Amount DECIMAL(12,2),
  Dept_ID INT,

);

Q: Create a WORKS_ON table with a composite primary key and two foreign keys
K: create table composite key many to many junction cascade
CREATE TABLE WORKS_ON (
  Essn CHAR(9) NOT NULL,
  Pno INT NOT NULL,
  Hours DECIMAL(3,1) NOT NULL DEFAULT 0.0,
  PRIMARY KEY (Essn, Pno),
  FOREIGN KEY (Essn) REFERENCES EMPLOYEE(Ssn) ON DELETE CASCADE,

Q: Create a procedure that raises an employee's salary by a tier that depends on the current salary
K: stored procedure if elseif else conditional declare update variable
DELIMITER $$
CREATE PROCEDURE ApplySalaryAdjustment(IN p_emp_id INT)
BEGIN
  DECLARE v_salary DECIMAL(10,2);
  SELECT Salary_Amount INTO v_salary FROM employee WHERE Employee_ID = p_emp_id;
  IF v_salary < 20000 THEN
    UPDATE employee SET Salary_Amount = v_salary * 1.15 WHERE Employee_ID = p_emp_id;
  ELSEIF v_salary < 40000 THEN
    UPDATE employee SET Salary_Amount = v_salary * 1.10 WHERE Employee_ID = p_emp_id;
  ELSE
    UPDATE employee SET Salary_Amount = v_salary * 1.05 WHERE Employee_ID = p_emp_id;
  END IF;
END $$
DELIMITER ;
CALL ApplySalaryAdjustment(118);

Q: Create a function that returns the salary band
K: function returns deterministic declare return
DELIMITER $$
CREATE FUNCTION Salary_Band(p_salary DECIMAL(10,2))
RETURNS VARCHAR(10)
DETERMINISTIC
BEGIN
  DECLARE v_band VARCHAR(10);
  IF p_salary < 30000 THEN SET v_band = 'Low';
  ELSE SET v_band = 'High';
  END IF;
  RETURN v_band;
END $$

Q: Create a trigger that logs every salary change into salary_audit_log
K: trigger after update audit log table old new history record change
CREATE TABLE IF NOT EXISTS salary_audit_log (
  Log_ID INT AUTO_INCREMENT PRIMARY KEY,
  Employee_ID INT,
  Old_Salary DECIMAL(10,2),
  New_Salary DECIMAL(10,2),
  Changed_At DATETIME
);
DELIMITER $$
CREATE TRIGGER trg_log_salary_change
AFTER UPDATE ON employee
FOR EACH ROW
BEGIN
  IF OLD.Salary_Amount <> NEW.Salary_Amount THEN
    INSERT INTO salary_audit_log (Employee_ID, Old_Salary, New_Salary, Changed_At)
    VALUES (OLD.Employee_ID, OLD.Salary_Amount, NEW.Salary_Amount, NOW());
  END IF;
END $$
DELIMITER ;

Q: Create a trigger that rejects a salary decrease
K: trigger before update signal sqlstate prevent block reject error message
DELIMITER $$
CREATE TRIGGER trg_no_pay_cut
BEFORE UPDATE ON employee
FOR EACH ROW
BEGIN
  IF NEW.Salary_Amount < OLD.Salary_Amount THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Salary cannot be decreased';
  END IF;
END $$
DELIMITER ;
