### 2
SELECT full_name, pay FROM staff WHERE pay BETWEEN 20000 AND 45000;
### 3

### 4

### 5

### 6

### 7
-- ORDERED
SELECT dish_name, SUM(price*qty) r FROM dish JOIN bill_item USING(dish_code) GROUP BY dish_code, dish_name ORDER BY r DESC;
### 8

### 9
SELECT bill_no, bill_date FROM bill WHERE YEAR(bill_date)=2024 AND tip IS NULL;
### 10

### 11

### 12

### 13

### 14

### 19

### 20
CREATE VIEW Outlet_Income AS SELECT o.title, SUM(d.price*i.qty) AS Income FROM outlet o JOIN bill b USING(outlet_no) JOIN bill_item i USING(bill_no) JOIN dish d USING(dish_code) GROUP BY o.outlet_no, o.title;
-- PROBE
SELECT title, Income FROM Outlet_Income WHERE Income IS NOT NULL;
